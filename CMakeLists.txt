cmake_minimum_required(VERSION 3.22)
project(Fader VERSION 0.2.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options($<$<COMPILE_LANG_AND_ID:CXX,MSVC>:/utf-8>)
option(FADER_BUILD_APP "Build native application and media tools" ON)
option(FADER_ENABLE_UI_FIXTURES "Enable synthetic UI qualification fixtures (no device)" OFF)
include(CTest)
set(FADER_JUCE_REVISION d60b7a78030661574a615b586d9325ba7a7fa7dd)
find_package(Git QUIET)
set(FADER_SOURCE_REVISION "source-archive")
if(GIT_FOUND AND EXISTS "${CMAKE_SOURCE_DIR}/.git")
    execute_process(COMMAND "${GIT_EXECUTABLE}" rev-parse HEAD
        WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}" OUTPUT_VARIABLE FADER_SOURCE_REVISION
        OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
endif()
configure_file(cmake/build-info.json.in fader-build-info.json @ONLY)

find_package(Threads REQUIRED)
add_library(fader_core src/Engine.cpp src/Playback.cpp src/Streaming.cpp src/PlaybackPreparation.cpp src/OfflineRender.cpp)
target_link_libraries(fader_core PUBLIC Threads::Threads)
target_include_directories(fader_core PUBLIC include)
target_compile_options(fader_core PRIVATE $<$<CXX_COMPILER_ID:Clang,AppleClang,GNU>:-Wall;-Wextra;-Wpedantic>)
if(BUILD_TESTING)
    add_executable(fader_core_tests tests/CoreTests.cpp tests/MixerTests.cpp tests/PlaybackTests.cpp)
    target_link_libraries(fader_core_tests PRIVATE fader_core Threads::Threads)
    add_test(NAME core COMMAND fader_core_tests)
    add_executable(fader_realtime_tests tests/RealtimeTests.cpp)
    target_link_libraries(fader_realtime_tests PRIVATE fader_core)
    add_test(NAME realtime_core COMMAND fader_realtime_tests)
    add_executable(fader_streaming_tests tests/StreamingTests.cpp)
    target_link_libraries(fader_streaming_tests PRIVATE fader_core)
    add_test(NAME streaming COMMAND fader_streaming_tests)
    add_executable(fader_seek_tests tests/SeekTests.cpp tests/PreparationTests.cpp)
    target_link_libraries(fader_seek_tests PRIVATE fader_core)
    add_test(NAME seeking COMMAND fader_seek_tests)
    add_executable(fader_offline_tests tests/OfflineTests.cpp)
    target_link_libraries(fader_offline_tests PRIVATE fader_core)
    add_test(NAME offline COMMAND fader_offline_tests)
    set_tests_properties(core realtime_core streaming seeking offline PROPERTIES TIMEOUT 60)
endif()

if(FADER_BUILD_APP)
    # JUCE 9.0.5. Allow an existing checkout for offline/repeated builds.
    set(FADER_JUCE_PATH "" CACHE PATH "Existing JUCE 9.0.5 checkout")
    if(FADER_JUCE_PATH)
        find_package(Git REQUIRED)
        execute_process(COMMAND "${GIT_EXECUTABLE}" rev-parse HEAD
            WORKING_DIRECTORY "${FADER_JUCE_PATH}" OUTPUT_VARIABLE actual_juce_revision
            OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
        if(NOT actual_juce_revision STREQUAL FADER_JUCE_REVISION)
            message(FATAL_ERROR "FADER_JUCE_PATH must be the pinned JUCE revision ${FADER_JUCE_REVISION}")
        endif()
        add_subdirectory("${FADER_JUCE_PATH}" "${CMAKE_BINARY_DIR}/juce")
    else()
        include(FetchContent)
        FetchContent_Declare(JUCE
            GIT_REPOSITORY https://github.com/juce-framework/JUCE.git
            GIT_TAG ${FADER_JUCE_REVISION})
        FetchContent_MakeAvailable(JUCE)
    endif()
    add_library(fader_media STATIC src/Media.cpp src/Devices.cpp src/Library.cpp src/AudioSetup.cpp
        src/AudioPlayback.cpp src/FileSource.cpp src/DecodedCache.cpp src/ProjectStorage.cpp src/ManifestJson.cpp src/WavExport.cpp)
    target_include_directories(fader_media PUBLIC src)
    target_compile_definitions(fader_media PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0
        JUCE_USE_MP3AUDIOFORMAT=1 JUCE_MODAL_LOOPS_PERMITTED=0)
    target_link_libraries(fader_media PUBLIC fader_core juce::juce_audio_utils juce::juce_cryptography)
    if(APPLE)
        target_link_libraries(fader_media PUBLIC "-framework IOKit")
    endif()
    juce_add_gui_app(Fader PRODUCT_NAME "Fader" COMPANY_NAME "Fader"
        BUNDLE_ID "org.fader.poc" NEEDS_CURL FALSE NEEDS_WEB_BROWSER FALSE)
    target_compile_definitions(Fader PRIVATE FADER_ENABLE_UI_FIXTURES=$<BOOL:${FADER_ENABLE_UI_FIXTURES}>)
    target_sources(Fader PRIVATE src/Main.cpp src/Arrangement.cpp src/AutomationLayer.cpp src/LibraryView.cpp)
    target_link_libraries(Fader PRIVATE fader_media)
    juce_add_console_app(fader_probe PRODUCT_NAME "fader_probe")
    target_sources(fader_probe PRIVATE src/Probe.cpp src/StreamQualification.cpp src/SeekBenchmark.cpp
        src/ExportQualification.cpp src/StressClock.cpp src/StressQualification.cpp src/StressFaults.cpp src/Arrangement.cpp src/AutomationLayer.cpp)
    target_link_libraries(fader_probe PRIVATE fader_media)
    if(BUILD_TESTING)
        juce_add_console_app(fader_media_tests PRODUCT_NAME "fader_media_tests")
        target_sources(fader_media_tests PRIVATE tests/MediaTests.cpp tests/ArrangementTests.cpp tests/AutomationTileTests.cpp
            tests/AudioSetupTests.cpp tests/SourceRateTests.cpp tests/StreamMediaTests.cpp tests/DecodedCacheTests.cpp tests/ProjectModelTests.cpp tests/ExportTests.cpp tests/PortabilityTests.cpp
            src/Arrangement.cpp src/AutomationLayer.cpp src/StreamQualification.cpp)
        target_compile_definitions(fader_media_tests PRIVATE
            FADER_TEST_MP3_PATH="${JUCE_SOURCE_DIR}/examples/Assets/Notifications/sounds/served.mp3"
            FADER_TEST_PROJECT_PATH="${CMAKE_SOURCE_DIR}/tests/fixtures/portable-v3.json"
            FADER_TEST_VBR_PATH="${CMAKE_SOURCE_DIR}/tests/fixtures/synthetic-vbr.mp3")
        target_link_libraries(fader_media_tests PRIVATE fader_media)
        add_test(NAME media COMMAND fader_media_tests)
        set_tests_properties(media PROPERTIES TIMEOUT 120)
        add_test(NAME software_stress_smoke COMMAND fader_probe --stress-self-test)
        set_tests_properties(software_stress_smoke PROPERTIES TIMEOUT 120)
    endif()
endif()
